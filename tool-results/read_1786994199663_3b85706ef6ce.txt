    14→    14→function waitForImages(container: HTMLDivElement): Promise<void> {
    15→    15→  const imgs = Array.from(container.querySelectorAll('img'));
    16→    16→  if (imgs.length === 0) return Promise.resolve();
    17→    17→  return Promise.all(
    18→    18→    imgs.map(
    19→    19→      (img) =>
    20→    20→        new Promise<void>((resolve) => {
    21→    21→          if (img.complete) {
    22→    22→            resolve();
    23→    23→          } else {
    24→    24→            img.onload = () => resolve();
    25→    25→            img.onerror = () => resolve();
    26→    26→            setTimeout(resolve, 3000);
    27→    27→          }
    28→    28→        })
    29→    29→    )
    30→    30→  ).then(() => {});
    31→    31→}

    33→    33→async function renderAndDownload(html: string, filename: string) {
    34→    34→  let container: HTMLDivElement | null = null;
    35→    35→  try {
    36→    36→    const jsPDF = (await import('jspdf')).default;
    37→    37→    const html2canvas = (await import('html2canvas')).default;
    38→    38→
    39→    39→    container = document.createElement('div');
    40→    40→    container.style.cssText = 'position:fixed;left:-9999px;top:0;width:700px;padding:24px;background:white;font-family:sans-serif;z-index:-1;';
    41→    41→    container.innerHTML = html;
    42→    42→    document.body.appendChild(container);
    43→    43→
    44→    44→    await waitForImages(container);
    45→    45→    await new Promise<void>((r) => setTimeout(r, 500));
    46→    46→
    47→    47→    const canvas = await html2canvas(container, {
    48→    48→      scale: 2,
    49→    49→      useCORS: true,
    50→    50→      allowTaint: true,
    51→    51→      logging: false,
    52→    52→    });
    53→    53→
    54→    54→    const pdf = new jsPDF('p', 'mm', 'a4');
    55→    55→    const imgWidth = 190;
    56→    56→    const pageHeight = 277;
    57→    57→    const imgHeight = (canvas.height * imgWidth) / canvas.width;
    58→    58→
    59→    59→    if (imgHeight <= pageHeight) {
    60→    60→      pdf.addImage(canvas.toDataURL('image/png'), 'PNG', 10, 10, imgWidth, imgHeight);
    61→    61→    } else {
    62→    62→      const totalPages = Math.ceil(imgHeight / pageHeight);
    63→    63→      for (let i = 0; i < totalPages; i++) {
    64→    64→        if (i > 0) pdf.addPage();
    65→    65→        const srcY = (i * pageHeight / imgHeight) * canvas.height;
    66→    66→        const srcH = Math.min((pageHeight / imgHeight) * canvas.height, canvas.height - srcY);
    67→    67→        const slice = document.createElement('canvas');
    68→    68→        slice.width = canvas.width;
    69→    69→        slice.height = srcH;
    70→    70→        const ctx = slice.getContext('2d');
    71→    71→        if (ctx) {
    72→    72→          ctx.drawImage(canvas, 0, srcY, canvas.width, srcH, 0, 0, canvas.width, srcH);
    73→    73→        }
    74→    74→        const sliceImgH = (srcH * imgWidth) / canvas.width;
    75→    75→        pdf.addImage(slice.toDataURL('image/png'), 'PNG', 10, 10, imgWidth, sliceImgH);
    76→    76→      }
    77→    77→    }
    78→    78→    pdf.save(filename);
    79→    79→  } catch (err) {
    80→    80→    console.error('PDF export error:', err);
    81→    81→    alert('Erro ao gerar PDF. Tente novamente.');
    82→    82→  } finally {
    83→    83→    if (container && container.parentNode) {
    84→    84→      container.parentNode.removeChild(container);
    85→    85→    }
    86→    86→  }
    87→    87→}
