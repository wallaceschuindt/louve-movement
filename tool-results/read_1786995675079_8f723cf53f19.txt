    14→    14→function waitForImages(container: HTMLDivElement): Promise<void> {
    15→    15→  const imgs = Array.from(container.querySelectorAll('img'));
    16→    16→  if (imgs.length === 0) return Promise.resolve();
    17→    17→  return Promise.all(
    18→    18→    imgs.map(
    19→    19→      (img) =>
    20→    20→        new Promise<void>((resolve) => {
    21→    21→          if (img.complete && img.naturalWidth > 0) {
    22→    22→            resolve();
    23→    23→          } else {
    24→    24→            const onDone = () => {
    25→    25→              img.removeEventListener('load', onDone);
    26→    26→              img.removeEventListener('error', onError);
    27→    27→              resolve();
    28→    28→            };
    29→    29→            const onError = () => {
    30→    30→              img.removeEventListener('load', onDone);
    31→    31→              img.removeEventListener('error', onError);
    32→    32→              img.style.display = 'none';
    33→    33→              const ph = document.createElement('div');
    34→    34→              ph.style.cssText = 'width:100%;height:100%;background:#e2e8f0;display:flex;align-items:center;justify-content:center;color:#94a3b8;font-size:10px;';
    35→    35→              ph.textContent = 'SEM IMAGEM';
    36→    36→              img.parentNode?.replaceChild(ph, img);
    37→    37→              resolve();
    38→    38→            };
    39→    39→            img.addEventListener('load', onDone);
    40→    40→            img.addEventListener('error', onError);
    41→    41→            setTimeout(onError, 4000);
    42→    42→          }
    43→    43→        })
    44→    44→    )
    45→    45→  ).then(() => {});
    46→    46→}

    48→    48→async function renderAndDownload(html: string, filename: string) {
    49→    49→  let container: HTMLDivElement | null = null;
    50→    50→  try {
    51→    51→    const jsPDF = (await import('jspdf')).default;
    52→    52→    const html2canvas = (await import('html2canvas')).default;
    53→    53→
    54→    54→    container = document.createElement('div');
    55→    55→    container.style.cssText = 'position:fixed;top:0;left:0;width:700px;padding:24px;background:white;font-family:sans-serif;z-index:-9999;opacity:0;pointer-events:none;overflow:visible;';
    56→    56→    container.innerHTML = html;
    57→    57→    document.body.appendChild(container);
    58→    58→
    59→    59→    await waitForImages(container);
    60→    60→    await new Promise<void>((r) => setTimeout(r, 800));
    61→    61→
    62→    62→    const canvas = await html2canvas(container, {
    63→    63→      scale: 2,
    64→    64→      useCORS: true,
    65→    65→      logging: false,
    66→    66→      windowWidth: 700,
    67→    67→      backgroundColor: '#ffffff',
    68→    68→    });
    69→    69→
    70→    70→    const pdf = new jsPDF('p', 'mm', 'a4');
    71→    71→    const imgWidth = 190;
    72→    72→    const pageHeight = 277;
    73→    73→    const imgHeight = (canvas.height * imgWidth) / canvas.width;
    74→    74→
    75→    75→    if (imgHeight <= pageHeight) {
    76→    76→      pdf.addImage(canvas.toDataURL('image/png'), 'PNG', 10, 10, imgWidth, imgHeight);
    77→    77→    } else {
    78→    78→      const totalPages = Math.ceil(imgHeight / pageHeight);
    79→    79→      for (let i = 0; i < totalPages; i++) {
    80→    80→        if (i > 0) pdf.addPage();
    81→    81→        const srcY = (i * pageHeight / imgHeight) * canvas.height;
    82→    82→        const srcH = Math.min((pageHeight / imgHeight) * canvas.height, canvas.height - srcY);
    83→    83→        const slice = document.createElement('canvas');
    84→    84→        slice.width = canvas.width;
    85→    85→        slice.height = srcH;
    86→    86→        const ctx = slice.getContext('2d');
    87→    87→        if (ctx) {
    88→    88→          ctx.drawImage(canvas, 0, srcY, canvas.width, srcH, 0, 0, canvas.width, srcH);
    89→    89→        }
    90→    90→        const sliceImgH = (srcH * imgWidth) / canvas.width;
    91→    91→        pdf.addImage(slice.toDataURL('image/png'), 'PNG', 10, 10, imgWidth, sliceImgH);
    92→    92→      }
    93→    93→    }
    94→    94→    pdf.save(filename);
    95→    95→  } catch (err) {
    96→    96→    console.error('PDF export error:', err);
    97→    97→    alert('Erro ao gerar PDF. Tente novamente.');
    98→    98→  } finally {
    99→    99→    if (container && container.parentNode) {
   100→   100→      container.parentNode.removeChild(container);
   101→   101→    }
   102→   102→  }
   103→   103→}
