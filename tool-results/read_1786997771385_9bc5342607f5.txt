    52→    52→    52→async function renderHTMLToCanvas(html: string): Promise<HTMLCanvasElement> {
    53→    53→    53→  // Renderiza dentro de um iframe isolado (documento em branco, sem o CSS
    54→    54→    54→  // global do app). Isso evita que a folha de estilo do Tailwind v4 (que usa
    55→    55→    55→  // cores no formato oklch() em regras como `* { border-color: ... }`)
    56→    56→    56→  // seja aplicada ao conteudo do PDF. O html2canvas 1.x nao sabe interpretar
    57→    57→    57→  // oklch() e, ao tentar, lancava um erro que interrompia a geracao do PDF
    58→    58→    58→  // (essa era a causa do "Erro ao gerar PDF").
    59→    59→    59→  const iframe = document.createElement('iframe');
    60→    60→    60→  iframe.style.cssText = 'position:fixed;left:-9999px;top:0;width:700px;height:0;border:0;visibility:hidden;';
    61→    61→    61→  document.body.appendChild(iframe);
    62→    62→    62→
    63→    63→    63→  try {
    64→    64→    64→    const doc = iframe.contentDocument;
    65→    65→    65→    if (!doc) throw new Error('Nao foi possivel criar area de renderizacao do PDF.');
    66→    66→    66→
    67→    67→    67→    doc.open();
    68→    68→    68→    doc.write('<!DOCTYPE html><html><head><meta charset="utf-8"></head><body></body></html>');
    69→    69→    69→    doc.close();
    70→    70→    70→
    71→    71→    71→    const container = doc.body;
    72→    72→    72→    container.style.cssText = 'margin:0;width:700px;background:#ffffff;font-family:Arial,Helvetica,sans-serif;color:#0f172a;';
    73→    73→    73→    container.innerHTML = html;
    74→    74→    74→
    75→    75→    75→    await waitForImages(container);
    76→    76→    76→    await new Promise<void>((r) => setTimeout(r, 500));
    77→    77→    77→
    78→    78→    78→    const canvas = await html2canvas(container, {
    79→    79→    79→      scale: 2,
    80→    80→    80→      useCORS: true,
    81→    81→    81→      allowTaint: true,
    82→    82→    82→      logging: false,
    83→    83→    83→      windowWidth: 700,
    84→    84→    84→      backgroundColor: '#ffffff',
    85→    85→    85→    });
    86→    86→    86→    return canvas;
    87→    87→    87→  } finally {
    88→    88→    88→    if (iframe.parentNode) {
    89→    89→    89→      iframe.parentNode.removeChild(iframe);
    90→    90→    90→    }
    91→    91→    91→  }
    92→    92→    92→}

    94→    94→    94→function canvasToPDF(canvas: HTMLCanvasElement): jsPDF {
    95→    95→    95→  const pdf = new jsPDF('p', 'mm', 'a4');
    96→    96→    96→  const imgWidth = 190;
    97→    97→    97→  const pageHeight = 277;
    98→    98→    98→  const imgHeight = (canvas.height * imgWidth) / canvas.width;
    99→    99→    99→
   100→   100→   100→  if (imgHeight <= pageHeight) {
   101→   101→   101→    pdf.addImage(canvas.toDataURL('image/png'), 'PNG', 10, 10, imgWidth, imgHeight);
   102→   102→   102→  } else {
   103→   103→   103→    const totalPages = Math.ceil(imgHeight / pageHeight);
   104→   104→   104→    for (let i = 0; i < totalPages; i++) {
   105→   105→   105→      if (i > 0) pdf.addPage();
   106→   106→   106→      const srcY = (i * pageHeight / imgHeight) * canvas.height;
   107→   107→   107→      const srcH = Math.min((pageHeight / imgHeight) * canvas.height, canvas.height - srcY);
   108→   108→   108→      const slice = document.createElement('canvas');
   109→   109→   109→      slice.width = canvas.width;
   110→   110→   110→      slice.height = srcH;
   111→   111→   111→      const ctx = slice.getContext('2d');
   112→   112→   112→      if (ctx) {
   113→   113→   113→        ctx.drawImage(canvas, 0, srcY, canvas.width, srcH, 0, 0, canvas.width, srcH);
   114→   114→   114→      }
   115→   115→   115→      const sliceImgH = (srcH * imgWidth) / canvas.width;
   116→   116→   116→      pdf.addImage(slice.toDataURL('image/png'), 'PNG', 10, 10, imgWidth, sliceImgH);
   117→   117→   117→    }
   118→   118→   118→  }
   119→   119→   119→  return pdf;
   120→   120→   120→}
