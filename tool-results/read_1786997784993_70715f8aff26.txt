    18→function waitForImages(container: HTMLElement): Promise<void> {
    19→  const imgs = Array.from(container.querySelectorAll('img'));
    20→  if (imgs.length === 0) return Promise.resolve();
    21→  return Promise.all(
    22→    imgs.map(
    23→      (img) =>
    24→        new Promise<void>((resolve) => {
    25→          if (img.complete && img.naturalWidth > 0) {
    26→            resolve();
    27→          } else {
    28→            const onDone = () => {
    29→              img.removeEventListener('load', onDone);
    30→              img.removeEventListener('error', onError);
    31→              resolve();
    32→            };
    33→            const onError = () => {
    34→              img.removeEventListener('load', onDone);
    35→              img.removeEventListener('error', onError);
    36→              img.style.display = 'none';
    37→              const ph = document.createElement('div');
    38→              ph.style.cssText = 'width:100%;height:100%;background:#e2e8f0;display:flex;align-items:center;justify-content:center;color:#94a3b8;font-size:10px;';
    39→              ph.textContent = 'SEM IMAGEM';
    40→              img.parentNode?.replaceChild(ph, img);
    41→              resolve();
    42→            };
    43→            img.addEventListener('load', onDone);
    44→            img.addEventListener('error', onError);
    45→            setTimeout(onError, 4000);
    46→          }
    47→        })
    48→    )
    49→  ).then(() => {});
    50→}

    52→async function renderHTMLToCanvas(html: string): Promise<HTMLCanvasElement> {
    53→  // Renderiza dentro de um iframe isolado (documento em branco, sem o CSS
    54→  // global do app). Isso evita que a folha de estilo do Tailwind v4 (que usa
    55→  // cores no formato oklch() em regras como `* { border-color: ... }`)
    56→  // seja aplicada ao conteudo do PDF. O html2canvas 1.x nao sabe interpretar
    57→  // oklch() e, ao tentar, lancava um erro que interrompia a geracao do PDF
    58→  // (essa era a causa do "Erro ao gerar PDF").
    59→  const iframe = document.createElement('iframe');
    60→  iframe.style.cssText = 'position:fixed;left:-9999px;top:0;width:700px;height:0;border:0;visibility:hidden;';
    61→  document.body.appendChild(iframe);
    62→
    63→  try {
    64→    const doc = iframe.contentDocument;
    65→    if (!doc) throw new Error('Nao foi possivel criar area de renderizacao do PDF.');
    66→
    67→    doc.open();
    68→    doc.write('<!DOCTYPE html><html><head><meta charset="utf-8"></head><body></body></html>');
    69→    doc.close();
    70→
    71→    const container = doc.body;
    72→    container.style.cssText = 'margin:0;width:700px;background:#ffffff;font-family:Arial,Helvetica,sans-serif;color:#0f172a;';
    73→    container.innerHTML = html;
    74→
    75→    await waitForImages(container);
    76→    await new Promise<void>((r) => setTimeout(r, 500));
    77→
    78→    const canvas = await html2canvas(container, {
    79→      scale: 2,
    80→      useCORS: true,
    81→      allowTaint: true,
    82→      logging: false,
    83→      windowWidth: 700,
    84→      backgroundColor: '#ffffff',
    85→    });
    86→    return canvas;
    87→  } finally {
    88→    if (iframe.parentNode) {
    89→      iframe.parentNode.removeChild(iframe);
    90→    }
    91→  }
    92→}

    94→function canvasToPDF(canvas: HTMLCanvasElement): jsPDF {
    95→  const pdf = new jsPDF('p', 'mm', 'a4');
    96→  const imgWidth = 190;
    97→  const pageHeight = 277;
    98→  const imgHeight = (canvas.height * imgWidth) / canvas.width;
    99→
   100→  if (imgHeight <= pageHeight) {
   101→    pdf.addImage(canvas.toDataURL('image/png'), 'PNG', 10, 10, imgWidth, imgHeight);
   102→  } else {
   103→    const totalPages = Math.ceil(imgHeight / pageHeight);
   104→    for (let i = 0; i < totalPages; i++) {
   105→      if (i > 0) pdf.addPage();
   106→      const srcY = (i * pageHeight / imgHeight) * canvas.height;
   107→      const srcH = Math.min((pageHeight / imgHeight) * canvas.height, canvas.height - srcY);
   108→      const slice = document.createElement('canvas');
   109→      slice.width = canvas.width;
   110→      slice.height = srcH;
   111→      const ctx = slice.getContext('2d');
   112→      if (ctx) {
   113→        ctx.drawImage(canvas, 0, srcY, canvas.width, srcH, 0, 0, canvas.width, srcH);
   114→      }
   115→      const sliceImgH = (srcH * imgWidth) / canvas.width;
   116→      pdf.addImage(slice.toDataURL('image/png'), 'PNG', 10, 10, imgWidth, sliceImgH);
   117→    }
   118→  }
   119→  return pdf;
   120→}

   122→async function renderAndDownload(html: string, filename: string) {
   123→  try {
   124→    const canvas = await renderHTMLToCanvas(html);
   125→    const pdf = canvasToPDF(canvas);
   126→    pdf.save(filename);
   127→  } catch (err) {
   128→    console.error('PDF export error:', err);
   129→    alert('Erro ao gerar PDF. Tente novamente.');
   130→  }
   131→}

   133→async function renderToBlob(html: string, filename: string): Promise<Blob | null> {
   134→  try {
   135→    const canvas = await renderHTMLToCanvas(html);
   136→    const pdf = canvasToPDF(canvas);
   137→    return pdf.output('blob');
   138→  } catch (err) {
   139→    console.error('PDF blob error:', err);
   140→    return null;
   141→  }

   144→function buildWhatsAppSummary(sale: SaleRecord | OtherSaleRecord, isOther: boolean): string {
   145→  const lines: string[] = [];
   146→  lines.push('*Romaneio ' + sale.id + '*');
   147→  lines.push('Data: ' + sale.date);
   148→  lines.push('');
   149→  lines.push('*Cliente:* ' + sale.client.name);
   150→  lines.push('*Pagamento:* ' + sale.paymentMethod);
   151→  lines.push('');
   152→  lines.push('*Itens:*');
   153→  if (!isOther) {
   154→    const s = sale as SaleRecord;
   155→    s.items.forEach((i) => {
   156→      lines.push('• ' + i.name + ' - ' + i.print + ' (' + i.color + ') | Tam. ' + i.size + ' | Qtd: ' + i.qty + ' | R$ ' + (i.price * i.qty).toFixed(2));
   157→    });
   158→  } else {
   159→    const s = sale as OtherSaleRecord;
   160→    s.items.forEach((i) => {
   161→      lines.push('• ' + i.name + ' - ' + i.category + ' | Qtd: ' + i.qty + ' | R$ ' + (i.price * i.qty).toFixed(2));
   162→    });
   163→  }
   164→  lines.push('');
   165→  lines.push('*Total: R$ ' + sale.total.toFixed(2) + '*');
   166→  return lines.join('\n');
   167→}

   169→export async function sharePDFWhatsApp(
   170→  settings: AppSettings,
   171→  sale: SaleRecord | OtherSaleRecord,
   172→  isOther: boolean
   173→) {
   174→  const html = buildRomaneioHtml(settings, sale, isOther);
   175→  const filename = 'Romaneio_' + sale.id + '_' + sale.client.name.replace(/\s+/g, '_') + '.pdf';
   176→
   177→  const blob = await renderToBlob(html, filename);
   178→  if (!blob) {
   179→    alert('Erro ao gerar PDF para WhatsApp. Tente exportar o PDF primeiro.');
   180→    return;
   181→  }
   182→
   183→  const phone = sale.client.phone.replace(/\D/g, '');
   184→  const summaryMsg = buildWhatsAppSummary(sale, isOther);
   185→
   186→  // Caminho ideal: dispositivos (principalmente celular) cujo navegador
   187→  // suporta compartilhar arquivos de verdade abrem direto o app do WhatsApp
   188→  // ja com o PDF bonito anexado, pronto para enviar.
   189→  if (navigator.share && navigator.canShare) {
   190→    const file = new File([blob], filename, { type: 'application/pdf' });
   191→    const shareData = {
   192→      title: 'Romaneio ' + sale.id,
   193→      text: summaryMsg,
   194→      files: [file],
   195→    };
   196→    if (navigator.canShare(shareData)) {
   197→      try {
   198→        await navigator.share(shareData);
   199→        return;
   200→      } catch (e) {
   201→        if ((e as DOMException).name === 'AbortError') return;
   202→      }
   203→    }
   204→  }
   205→
   206→  // Fallback (principalmente navegadores de computador, que ainda nao
   207→  // permitem que um site anexe um arquivo automaticamente ao WhatsApp Web):
   208→  // baixamos o PDF de verdade e abrimos o WhatsApp com o romaneio completo
   209→  // em texto, para o usuario so precisar arrastar o PDF ja baixado.
   210→  const url = URL.createObjectURL(blob);
   211→  const a = document.createElement('a');
   212→  a.href = url;
   213→  a.download = filename;
   214→  a.click();
   215→  setTimeout(() => URL.revokeObjectURL(url), 10000);
   216→
   217→  const waMsg = summaryMsg + '\n\n_O PDF (' + filename + ') foi baixado no seu computador. Anexe o arquivo aqui antes de enviar._';
   218→  const waUrl = phone
   219→    ? 'https://wa.me/' + phone + '?text=' + encodeURIComponent(waMsg)
   220→    : 'https://wa.me/?text=' + encodeURIComponent(waMsg);
   221→  window.open(waUrl, '_blank');
   222→}
