     9→function imgToHtml(src: string, w: number, h: number, radius = 8): string {
    10→  if (!src) return '<div style="width:' + w + 'px;height:' + h + 'px;border-radius:' + radius + 'px;background:#e2e8f0;display:flex;align-items:center;justify-content:center;color:#94a3b8;font-size:' + Math.floor(w / 3) + 'px;">SEM</div>';
    11→  return '<img src="' + src + '" style="width:' + w + 'px;height:' + h + 'px;border-radius:' + radius + 'px;object-fit:cover;" />';
    12→}

    14→function waitForImages(container: HTMLDivElement): Promise<void> {
    15→  const imgs = Array.from(container.querySelectorAll('img'));
    16→  if (imgs.length === 0) return Promise.resolve();
    17→  return Promise.all(
    18→    imgs.map(
    19→      (img) =>
    20→        new Promise<void>((resolve) => {
    21→          if (img.complete && img.naturalWidth > 0) {
    22→            resolve();
    23→          } else {
    24→            const onDone = () => {
    25→              img.removeEventListener('load', onDone);
    26→              img.removeEventListener('error', onError);
    27→              resolve();
    28→            };
    29→            const onError = () => {
    30→              img.removeEventListener('load', onDone);
    31→              img.removeEventListener('error', onError);
    32→              img.style.display = 'none';
    33→              const ph = document.createElement('div');
    34→              ph.style.cssText = 'width:100%;height:100%;background:#e2e8f0;display:flex;align-items:center;justify-content:center;color:#94a3b8;font-size:10px;';
    35→              ph.textContent = 'SEM IMAGEM';
    36→              img.parentNode?.replaceChild(ph, img);
    37→              resolve();
    38→            };
    39→            img.addEventListener('load', onDone);
    40→            img.addEventListener('error', onError);
    41→            setTimeout(onError, 4000);
    42→          }
    43→        })
    44→    )
    45→  ).then(() => {});
    46→}

    48→async function renderAndDownload(html: string, filename: string) {
    49→  let container: HTMLDivElement | null = null;
    50→  try {
    51→    const jsPDF = (await import('jspdf')).default;
    52→    const html2canvas = (await import('html2canvas')).default;
    53→
    54→    container = document.createElement('div');
    55→    container.style.cssText = 'position:fixed;top:0;left:0;width:700px;padding:24px;background:white;font-family:sans-serif;z-index:-9999;opacity:0;pointer-events:none;overflow:visible;';
    56→    container.innerHTML = html;
    57→    document.body.appendChild(container);
    58→
    59→    await waitForImages(container);
    60→    await new Promise<void>((r) => setTimeout(r, 800));
    61→
    62→    const canvas = await html2canvas(container, {
    63→      scale: 2,
    64→      useCORS: true,
    65→      logging: false,
    66→      windowWidth: 700,
    67→      backgroundColor: '#ffffff',
    68→    });
    69→
    70→    const pdf = new jsPDF('p', 'mm', 'a4');
    71→    const imgWidth = 190;
    72→    const pageHeight = 277;
    73→    const imgHeight = (canvas.height * imgWidth) / canvas.width;
    74→
    75→    if (imgHeight <= pageHeight) {
    76→      pdf.addImage(canvas.toDataURL('image/png'), 'PNG', 10, 10, imgWidth, imgHeight);
    77→    } else {
    78→      const totalPages = Math.ceil(imgHeight / pageHeight);
    79→      for (let i = 0; i < totalPages; i++) {
    80→        if (i > 0) pdf.addPage();
    81→        const srcY = (i * pageHeight / imgHeight) * canvas.height;
    82→        const srcH = Math.min((pageHeight / imgHeight) * canvas.height, canvas.height - srcY);
    83→        const slice = document.createElement('canvas');
    84→        slice.width = canvas.width;
    85→        slice.height = srcH;
    86→        const ctx = slice.getContext('2d');
    87→        if (ctx) {
    88→          ctx.drawImage(canvas, 0, srcY, canvas.width, srcH, 0, 0, canvas.width, srcH);
    89→        }
    90→        const sliceImgH = (srcH * imgWidth) / canvas.width;
    91→        pdf.addImage(slice.toDataURL('image/png'), 'PNG', 10, 10, imgWidth, sliceImgH);
    92→      }
    93→    }
    94→    pdf.save(filename);
    95→  } catch (err) {
    96→    console.error('PDF export error:', err);
    97→    alert('Erro ao gerar PDF. Tente novamente.');
    98→  } finally {
    99→    if (container && container.parentNode) {
   100→      container.parentNode.removeChild(container);
   101→    }
   102→  }
   103→}
