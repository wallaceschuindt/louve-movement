   144→export async function sharePDFWhatsApp(
   145→  settings: AppSettings,
   146→  sale: SaleRecord | OtherSaleRecord,
   147→  isOther: boolean
   148→) {
   149→  const html = buildRomaneioHtml(settings, sale, isOther);
   150→  const filename = 'Romaneio_' + sale.id + '_' + sale.client.name.replace(/\s+/g, '_') + '.pdf';
   151→
   152→  const blob = await renderToBlob(html, filename);
   153→  if (!blob) {
   154→    alert('Erro ao gerar PDF para WhatsApp. Tente exportar o PDF primeiro.');
   155→    return;
   156→  }
   157→
   158→  const phone = sale.client.phone.replace(/\D/g, '');
   159→  const baseMsg = 'Ola ' + sale.client.name + '! Segue seu romaneio ' + sale.id + ' - Total: R$ ' + sale.total.toFixed(2);
   160→
   161→  if (navigator.share && navigator.canShare) {
   162→    const file = new File([blob], filename, { type: 'application/pdf' });
   163→    const shareData = {
   164→      title: 'Romaneio ' + sale.id,
   165→      text: baseMsg,
   166→      files: [file],
   167→    };
   168→    if (navigator.canShare(shareData)) {
   169→      try {
   170→        await navigator.share(shareData);
   171→        return;
   172→      } catch (e) {
   173→        if ((e as DOMException).name === 'AbortError') return;
   174→      }
   175→    }
   176→  }
   177→
   178→  const url = URL.createObjectURL(blob);
   179→  const a = document.createElement('a');
   180→  a.href = url;
   181→  a.download = filename;
   182→  a.click();
   183→  setTimeout(() => URL.revokeObjectURL(url), 10000);
   184→
   185→  const waMsg = baseMsg + '\n\n*PDF do romaneio foi salvo no seu dispositivo. Por favor, anexe o arquivo PDF ao enviar esta mensagem.*';
   186→  const waUrl = phone
   187→    ? 'https://wa.me/' + phone + '?text=' + encodeURIComponent(waMsg)
   188→    : 'https://wa.me/?text=' + encodeURIComponent(waMsg);
   189→  window.open(waUrl, '_blank');
   190→}
